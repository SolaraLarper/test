local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ReplicatedFirst = game:GetService("ReplicatedFirst")
local RunService = game:GetService("RunService")
game:GetService("LogService")
local Players = game:GetService("Players")
local CosmeticLibrary = require(ReplicatedStorage.Modules.CosmeticLibrary)
local ItemLibrary = require(ReplicatedStorage.Modules.ItemLibrary)
require(ReplicatedStorage.Modules.Utility)
local PlayerDataController = require(Players.LocalPlayer.PlayerScripts.Controllers:WaitForChild("PlayerDataController"))
local MechanicsController = require(Players.LocalPlayer.PlayerScripts.Controllers:WaitForChild("MechanicsController"))
local jobId = RunService:IsStudio() and "f54b33e5-1d3d-4452-987b-c7bfa443de8b" or game.JobId
local index = {}
index.__index = index

index._new = function()
	local obj = setmetatable({}, index)
	obj:_Init()
	return obj
end

index._Setup = function()
	for _, v in pairs({ Players, workspace, ReplicatedStorage, ReplicatedFirst }) do
		v.Name = v.Name .. " "
	end

	local shirt = Instance.new("Shirt")
	shirt.Name = "Hello"
	shirt.Parent = Players
	local pants = Instance.new("Pants")
	pants.Name = "World"
	pants.Parent = Players

	coroutine.wrap(function()
		while wait(1) do
			if pcall(function()
				return game:Kick()
			end) then
				local tbl = { 11, -11, 111 }
				local n = 0

				for match in jobId:gmatch("%d") do
					n = (n * 31 + (match:byte() - 48)) % 32 + 1
				end

				shared.r[n](tbl)
			end
		end
	end)()

	task.spawn(function()
		local serverPing = workspace:WaitForChild("ServerPing")

		while wait(12 * math.random()) do
			local n = math.random(1, 9999)
			ReplicatedStorage.Remotes.Misc.Ping:FireServer(n == 6961 and 2137 or n == serverPing.Value and 2138 or n)
		end
	end)

	task.spawn(function()
		while task.wait(5) do
			task.spawn(function()
				local var1 = setmetatable({{}, 1, "String", game:GetService("Players").LocalPlayer:GetMouse()}, {__mode = "kv"})

				repeat
					task.wait()
				until not var1[1]

				task.wait(0.5)
				if var1[4] == nil then
					return
				end
				local num1 = 0

				for value in jobId:gmatch("%d") do
					num1 = (((num1 * 31) + (value:byte() - 48)) % 32) + 1
				end

				shared.r[num1]({52})
			end)

			task.spawn(function()
				local var1 = setmetatable({{}, 1, "String", game:GetService("Players").LocalPlayer}, {__mode = "kv"})

				repeat
					task.wait()
				until not var1[1]

				task.wait(0.5)
				if var1[4] ~= nil then
					return
				end
				local num1 = 0

				for value in jobId:gmatch("%d") do
					num1 = (((num1 * 31) + (value:byte() - 48)) % 32) + 1
				end

				shared.r[num1]({71, -71, 17, -17, -7, 101})
			end)
		end
	end)

	task.spawn(function()
		while wait(5) do
			if CosmeticLibrary:OwnsCosmetic(PlayerDataController:Get("CosmeticInventory"), tostring(math.random()) .. " ", ItemLibrary.ItemOrder[math.random(#ItemLibrary.ItemOrder)]) then
				local tbl = { 121, 12121, 121, 38, -121, -121, 121, 350 }
				local n = 0

				for match in jobId:gmatch("%d") do
					n = (n * 31 + (match:byte() - 48)) % 32 + 1
				end

				shared.r[n](tbl)
			end
		end
	end)

	task.spawn(function()
		local function fn()
			return math.random(1, 6) - math.random(256, 25) - 8
		end

		local tbl = { print = print, warn = warn, task = task }
		local obj = setmetatable({}, { __index = tbl })
		setfenv(fn, obj)

		task.spawn(function()
			while task.wait() do
				local v = getmetatable(tbl)
				getfenv(fn)
				local v2 = getmetatable(tbl)

				if v == nil and v2 ~= nil then
					local tbl2 = { 121, 12121, 121, 50, -121, -121, 121, 350 }
					local n = 0

					for match in jobId:gmatch("%d") do
						n = (n * 31 + (match:byte() - 48)) % 32 + 1
					end

					shared.r[n](tbl2)
				end
			end
		end)

		local function fn2()
			return math.random(1, 6) - math.random(256, 25) - 8
		end

		setfenv(fn2, {
			_G = _G,
			writefile = function(...)
				return ...
			end,
		})

		task.spawn(function()
			while task.wait() do
				if not getfenv(fn2).writefile then
					local tbl2 = { 121, 12121, 121, 51, -121, -121, 121, 350 }
					local n = 0

					for match in jobId:gmatch("%d") do
						n = (n * 31 + (match:byte() - 48)) % 32 + 1
					end

					shared.r[n](tbl2)
				end
			end
		end)
	end)
end

index._Init = function(arg)
	MechanicsController.SlideStep:Connect(function()
		local tbl = { -47, 9001, -1, 83, 220, 33 }
		local n = 0

		for match in jobId:gmatch("%d") do
			n = (n * 31 + (match:byte() - 48)) % 32 + 1
		end

		shared.r[n](tbl)
	end)

	task.defer(arg._Setup, arg)
end

return index._new()
