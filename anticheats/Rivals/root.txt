local n = 0

local function fn()

local function fn2(descendant)
	if (descendant:IsA("RemoteEvent") or descendant:IsA("UnreliableRemoteEvent")) and not descendant:GetAttribute("DontFlush") then
		table.insert(tbl, descendant.OnClientEvent:Connect(fn))
		return
	end
end

local function fn3()
	table.insert(tbl, remotes.DescendantAdded:Connect(fn2))

	for _, descendant in pairs(remotes:GetDescendants()) do
		task.defer(fn2, descendant)
	end
end

local function fn4()
	for _, v in pairs(tbl) do
		v:Disconnect()
	end

	tbl = {}
end

	StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.Backpack, false)

	RunService:UnbindFromRenderStep("LoadingScreen")
	TeleportService:SetTeleportSetting("SkipLoadingScreen", true)
	LoadingScreen:Close()
end
